## Step 2: Write `test/host/test_ringbuf.c`

New file:
```c
#include "test_runner.h"
#include "ringbuf.h"

static uint8_t store[16];

static void test_init_rejects(void)
{
  ringbuf_t rb;
  CHECK(!ringbuf_init(&rb, store, 0));
  CHECK(!ringbuf_init(&rb, store, 3));
  CHECK(!ringbuf_init(&rb, store, 12));
  CHECK(!ringbuf_init(&rb, NULL, 16));
  CHECK(!ringbuf_init(NULL, store, 16));
  CHECK(ringbuf_init(&rb, store, 16));
  CHECK(rb.mask == 15);
}

static void test_static_init(void)
{
  ringbuf_t rb = RINGBUF_STATIC_INIT(store, sizeof store);
  CHECK(rb.size == 16 && rb.mask == 15);
  CHECK(ringbuf_used(&rb) == 0);
  CHECK(ringbuf_free(&rb) == 16);
}

static void test_empty(void)
{
  ringbuf_t rb;
  const uint8_t *p = NULL;
  ringbuf_init(&rb, store, 16);
  CHECK(ringbuf_used(&rb) == 0);
  CHECK(ringbuf_free(&rb) == 16);
  CHECK(ringbuf_peek(&rb, &p) == 0);
  CHECK(ringbuf_write(&rb, (const uint8_t *)"x", 0) == 0);
  CHECK(rb.dropped == 0);             /* len 0 is not a drop */
}

static void test_full_and_drop(void)
{
  ringbuf_t rb;
  uint8_t d[16] = {0};
  ringbuf_init(&rb, store, 16);
  CHECK(ringbuf_write(&rb, d, 16) == 16);
  CHECK(ringbuf_used(&rb) == 16);
  CHECK(ringbuf_free(&rb) == 0);
  CHECK(ringbuf_write(&rb, d, 1) == 0);
  CHECK(rb.dropped == 1);
  CHECK(ringbuf_used(&rb) == 16);     /* full ring unchanged */
}

static void test_all_or_nothing(void)
{
  ringbuf_t rb;
  uint8_t d[16] = {0};
  ringbuf_init(&rb, store, 16);
  CHECK(ringbuf_write(&rb, d, 10) == 10);
  CHECK(ringbuf_write(&rb, d, 7) == 0);   /* free is 6 */
  CHECK(rb.dropped == 7);
  CHECK(ringbuf_used(&rb) == 10);         /* no partial write */
  CHECK(ringbuf_write(&rb, d, 6) == 6);
  CHECK(ringbuf_free(&rb) == 0);
}

/* head/tail near 2^32 AND index near buffer end: write must split,
 * peek must stop at buffer end, counters must wrap through 0. */
static void test_wrap_2pow32(void)
{
  ringbuf_t rb;
  const uint8_t *p;
  uint8_t d[10];
  for (int i = 0; i < 10; i++) d[i] = (uint8_t)i;

  ringbuf_init(&rb, store, 16);
  rb.head = rb.tail = 0xFFFFFFFAu;        /* idx = 10, 6 bytes before end */

  CHECK(ringbuf_write(&rb, d, 10) == 10);
  CHECK(rb.head == 4u);                   /* wrapped past 2^32 */
  CHECK(ringbuf_used(&rb) == 10);
  CHECK(ringbuf_free(&rb) == 6);

  CHECK(ringbuf_peek(&rb, &p) == 6);      /* stops at buffer end */
  CHECK(p == &store[10]);
  CHECK(memcmp(p, d, 6) == 0);
  ringbuf_consume(&rb, 6);

  CHECK(ringbuf_peek(&rb, &p) == 4);      /* rest from buffer start */
  CHECK(p == &store[0]);
  CHECK(memcmp(p, d + 6, 4) == 0);
  ringbuf_consume(&rb, 4);

  CHECK(ringbuf_used(&rb) == 0);
  CHECK(rb.tail == 4u);
}

static void test_consume_clamp(void)
{
  ringbuf_t rb;
  uint8_t d[3] = {1, 2, 3};
  ringbuf_init(&rb, store, 16);
  ringbuf_write(&rb, d, 3);
  ringbuf_consume(&rb, 100);
  CHECK(ringbuf_used(&rb) == 0);
  CHECK(rb.tail == rb.head);
}

int main(void)
{
  test_init_rejects();
  test_static_init();
  test_empty();
  test_full_and_drop();
  test_all_or_nothing();
  test_wrap_2pow32();
  test_consume_clamp();
  TEST_SUMMARY();
}
```

What each test covers:
| Test | Checks |
|---|---|
| `test_init_rejects` | size 0, 3, 12 (not a power of 2), NULL buf, and NULL rb are rejected; 16 is accepted and mask = 15 |
| `test_static_init` | `RINGBUF_STATIC_INIT` gives a valid empty ring with no init call |
| `test_empty` | used 0, free 16, peek returns 0; a write with len 0 is not counted as a drop |
| `test_full_and_drop` | 16 bytes fill the ring; 1 more byte is rejected, dropped = 1, and the ring is unchanged |
| `test_all_or_nothing` | with 6 bytes free, a 7-byte write is rejected in full (no partial write) and dropped = 7 |
| `test_wrap_2pow32` | head/tail preset to `0xFFFFFFFA` (index 10 of 16), so one 10-byte write crosses the buffer end **and** 2^32 at the same time; peek returns 6 bytes at `store[10]`, then 4 at `store[0]`; head and tail end at 4 |
| `test_consume_clamp` | `consume(100)` with 3 bytes used is clamped to 3 |

Why `0xFFFFFFFA`: `0xFFFFFFFA & 15 = 10`, so 6 bytes fit before the buffer end, and `0xFFFFFFFA + 10` wraps to `4`. That one write tests the split `memcpy` and the unsigned `head - tail` math across the 2^32 wrap. Real firmware would take days of streaming to get there.

```make
# Host unit tests (MinGW gcc). Run from test/host:  mingw32-make
# STM32CubeCLT puts sh.exe on PATH; force cmd so .\test_x works.
SHELL = cmd.exe

CFLAGS = -std=c11 -Wall -Wextra -g -I../../Lib/Inc -I.

TESTS = test_ringbuf

all: run

# cmd.exe is the shell under mingw32-make; && chain stops on first failure.

test_ringbuf: test_ringbuf.c ../../Lib/Src/ringbuf.c
	$(CC) $(CFLAGS) -DHOST_TEST $^ -o $@

clean:
	cmd /c "del /Q *.exe 2>nul & exit 0"
```
`ringbuf.c` has no HAL dependency, so it compiles on the host unchanged. `__atomic_thread_fence` (chosen on 2026-11-26 instead of `__DMB`) builds with MinGW gcc 6.3.

### Problem 1: TAB vs spaces

`cat -A Makefile` shows a tab as `^I`. In the first version of the file:
- line 2 `CC = gcc` started with a TAB by mistake,
- the 3 recipe lines (`@$(foreach ...`, `$(CC) $(CFLAGS) ...`, `cmd /c "del ...`) started with 8 spaces.

With spaces, make fails with `*** missing separator.  Stop.`

TAB is the `Tab` key, and it puts one tab character in the file. Spaces are different characters, even when they look the same on screen. To get real tabs in VS Code, click `Spaces: 4` in the bottom-right status bar and choose **Indent Using Tabs**.

Fix: remove the tab before `CC`, and start each recipe line with one TAB.

To check, from `test/host`:
```powershell
Select-String -Path Makefile -Pattern "^`t" | ForEach-Object { $_.LineNumber }
```
It should print only the recipe line numbers.

### Problem 2: `.test_ringbuf: not found` (Error 127)

After the tab fix, the build compiled but the run failed:
```
c:/st/stm32cubeclt_1.23.0/make/bin/sh: .test_ringbuf: not found
mingw32-make: *** [run] Error 127
```
Cause: STM32CubeCLT puts `C:\ST\STM32CubeCLT_1.23.0\Make\bin\sh.exe` on PATH. When `mingw32-make` finds a `sh.exe`, it uses it instead of `cmd.exe`, and sh treats the `\` in `.\test_ringbuf` as an escape, which leaves `.test_ringbuf`.
Fix: add `SHELL = cmd.exe` near the top of the Makefile (shown above). This forces cmd whatever is on PATH.

## Step 6: `flash.ps1` (build + flash in one command)

Before this, flashing took two commands (from the 2026-11-24 log):
```powershell
cmake --build --preset Debug
STM32_Programmer_CLI -c port=SWD -w build\Debug\sensor_fusion.elf -v -rst
```
`flash.ps1` wraps them and stops at the first error, so an old ELF is never flashed after a failed build. It is modelled on `F:\5kg_v2_firmware\flash.ps1`, without the calibration save/restore, because this project has no persistent config in flash yet.

New file `F:\sensor_fusion\flash.ps1` (project root, next to `CMakeLists.txt`):
```powershell
# ******************************************************************************
#  flash.ps1 - build + flash sensor_fusion to the NUCLEO-F767ZI over ST-Link (CN1)
#
#  Usage:
#     .\flash.ps1              # build Debug, flash, verify, reset
#     .\flash.ps1 -Release     # same, Release preset
#     .\flash.ps1 -NoBuild     # flash existing ELF only
#     .\flash.ps1 -Test        # run host tests (test/host) first; abort on failure
#     .\flash.ps1 -Erase       # full chip erase before programming
# ******************************************************************************
param(
  [switch]$Release,
  [switch]$NoBuild,
  [switch]$Test,
  [switch]$Erase
)

$ErrorActionPreference = "Stop"
Set-Location -Path $PSScriptRoot

$Preset     = if ($Release) { "Release" } else { "Debug" }
$BuildDir   = "build\$Preset"
$Elf        = "$BuildDir\sensor_fusion.elf"
$Programmer = "C:\ST\STM32CubeCLT_1.23.0\STM32CubeProgrammer\bin\STM32_Programmer_CLI.exe"
$MinGW      = "C:\MinGW\bin"
$Conn       = @("-c", "port=SWD", "freq=4000")

if (-not (Test-Path $Programmer)) { throw "STM32_Programmer_CLI not found: $Programmer" }

# --- Host tests (optional) ----------------------------------------------------
if ($Test) {
  Write-Host "==> Host tests..." -ForegroundColor Cyan
  $oldPath = $env:Path
  $env:Path = "$MinGW;$env:Path"
  Push-Location test\host
  try {
    & "$MinGW\mingw32-make.exe"
    if ($LASTEXITCODE -ne 0) { throw "Host tests FAILED - not flashing" }
  } finally {
    Pop-Location
    $env:Path = $oldPath
  }
}

# --- Build --------------------------------------------------------------------
if (-not $NoBuild) {
  # Configure only when the build dir is missing (first run or after deleting build\)
  if (-not (Test-Path "$BuildDir\build.ninja")) {
    Write-Host "==> Configuring ($Preset)..." -ForegroundColor Cyan
    cmake --preset $Preset
    if ($LASTEXITCODE -ne 0) { throw "CMake configure failed (exit $LASTEXITCODE)" }
  }
  Write-Host "==> Building ($Preset)..." -ForegroundColor Cyan
  cmake --build --preset $Preset
  if ($LASTEXITCODE -ne 0) { throw "Build failed (exit $LASTEXITCODE)" }
}

if (-not (Test-Path $Elf)) { throw "Firmware not found: $Elf  (run without -NoBuild first)" }

# --- Erase (optional) ---------------------------------------------------------
if ($Erase) {
  Write-Host "==> Full chip erase..." -ForegroundColor Yellow
  & $Programmer @Conn -e all
  if ($LASTEXITCODE -ne 0) { throw "Erase failed (exit $LASTEXITCODE)" }
}

# --- Flash + verify + reset ---------------------------------------------------
Write-Host "==> Flashing $Elf ..." -ForegroundColor Cyan
& $Programmer @Conn -w $Elf -v -rst
if ($LASTEXITCODE -ne 0) { throw "Flash failed (exit $LASTEXITCODE) - board on CN1? another tool holding ST-Link?" }

Write-Host "==> Done. Firmware flashed and running." -ForegroundColor Green
```

How it works:
- `param(...)`: command-line switches. `[switch]` means on or off, with no value.
- `$ErrorActionPreference = "Stop"`: a PowerShell error stops the script. External programs (cmake, the programmer) do not stop it by themselves, so each one is followed by an `if ($LASTEXITCODE -ne 0) { throw ... }` check.
- `Set-Location $PSScriptRoot`: the script works from any current folder.
- `@Conn` (splatting): passes the array as separate arguments, so `-c port=SWD freq=4000` is written only once. `freq=4000` is a 4 MHz SWD clock, the maximum for the on-board ST-Link V2-1.
- Configure runs only when `build\<preset>\build.ninja` is missing. After a CubeMX regenerate or a `CMakeLists.txt` edit, Ninja re-runs CMake by itself.
- `-Test` runs the `test/host` suite first and does not flash if it fails. `try/finally` restores PATH and the folder.
- `-w $Elf -v -rst`: write, verify by reading back, then reset. The ELF carries its own addresses, so none are passed. Only the sectors the ELF covers are erased.
- No calibration save/restore yet. Add it if config is ever stored in a flash sector (F7 sector trap: 32/128/256 KB sectors).
