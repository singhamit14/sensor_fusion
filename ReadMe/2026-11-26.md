void app_init(void)

}

void app_loop(void)
{
  uint8_t rx[64];

  /* echo: whatever the PC sends comes back */
  uint32_t len = usb_read(rx, sizeof(rx));
  if (len > 0)
  {
    if (rx[0] == 'b')       /* burst test: 100 lines, about 4 KB */
    {
      for (uint32_t i = 0; i < 100; i++)
        usb_printf("burst %03lu  0123456789abcdef\r\n", i);
    }
    else
    {
      usb_printf("echo: %.*s\r\n", (int)len, rx);
    }
  }

  /* every 500 ms: blink + tick */
  if (HAL_GetTick() - tick_last >= 500)
  {
    tick_last = HAL_GetTick();
    HAL_GPIO_TogglePin(LD1_GPIO_Port, LD1_Pin);
    usb_printf("tick %lu  ms=%lu  pi=%.3f  drop=%lu  free=%lu\r\n",
               tick_n, HAL_GetTick(), 3.14159f, usb_tx_dropped(), usb_tx_free());
    tick_n++;
  }
}

## Step 4: Move the ring logic into `Lib/ringbuf.c`

Idea: `Lib/ringbuf` is a generic single-producer / single-consumer byte ring (head/tail/drop counter, wrap math) with **no HAL**, so it can be host-tested. `usbd_cdc_if.c` keeps only the USB glue: `tx_inflight`, `usb_tx_kick()`, and the `OTG_FS_IRQn` masking.
`__DMB()` is CMSIS (not available on the PC), so `Lib` uses `__atomic_thread_fence(__ATOMIC_SEQ_CST)`. GCC emits `dmb` for it on Cortex-M7, and it also compiles with MinGW.

New file `Lib/Inc/ringbuf.h`:
```c
#ifndef RINGBUF_H
#define RINGBUF_H

#include <stdint.h>
#include <stdbool.h>

/* Single-producer / single-consumer byte ring.
 * head: written only by producer. tail: written only by consumer.
 * head/tail free-run (wrap at 2^32); index = value & mask.
 * size must be a power of 2. */
typedef struct
{
  uint8_t          *buf;
  uint32_t          size;
  uint32_t          mask;
  volatile uint32_t head;
  volatile uint32_t tail;
  volatile uint32_t dropped;   /* bytes rejected because ring was full */
} ringbuf_t;

/* Compile-time init, so ring is valid before any init code runs */
#define RINGBUF_STATIC_INIT(storage, sz) \
  { (storage), (sz), (sz) - 1u, 0u, 0u, 0u }

bool     ringbuf_init(ringbuf_t *rb, uint8_t *buf, uint32_t size);
uint32_t ringbuf_used(const ringbuf_t *rb);
uint32_t ringbuf_free(const ringbuf_t *rb);

/* Producer: all-or-nothing. Returns len, or 0 (and adds len to dropped). */
uint32_t ringbuf_write(ringbuf_t *rb, const uint8_t *data, uint32_t len);

/* Consumer: pointer + length of contiguous readable bytes at tail (stops at buffer end). */
uint32_t ringbuf_peek(const ringbuf_t *rb, const uint8_t **ptr);

/* Consumer: free n bytes (clamped to used). */
void     ringbuf_consume(ringbuf_t *rb, uint32_t n);

#endif /* RINGBUF_H */
```

New file `Lib/Src/ringbuf.c`:
```c
#include "ringbuf.h"
#include <string.h>

#define RB_FENCE()  __atomic_thread_fence(__ATOMIC_SEQ_CST)   /* dmb on Cortex-M7 */

bool ringbuf_init(ringbuf_t *rb, uint8_t *buf, uint32_t size)
{
  if (rb == NULL || buf == NULL) return false;
  if (size == 0 || (size & (size - 1u)) != 0) return false;   /* power of 2 */
  rb->buf     = buf;
  rb->size    = size;
  rb->mask    = size - 1u;
  rb->head    = 0;
  rb->tail    = 0;
  rb->dropped = 0;
  return true;
}

uint32_t ringbuf_used(const ringbuf_t *rb)
{
  return rb->head - rb->tail;
}

uint32_t ringbuf_free(const ringbuf_t *rb)
{
  return rb->size - (rb->head - rb->tail);
}

uint32_t ringbuf_write(ringbuf_t *rb, const uint8_t *data, uint32_t len)
{
  if (len == 0) return 0;
  if (len > ringbuf_free(rb))
  {
    rb->dropped += len;
    return 0;
  }

  uint32_t head  = rb->head;
  uint32_t idx   = head & rb->mask;
  uint32_t first = rb->size - idx;        /* space before wrap */
  if (first > len) first = len;
  memcpy(&rb->buf[idx], data, first);
  memcpy(&rb->buf[0], data + first, len - first);

  RB_FENCE();                             /* data in ring before head moves */
  rb->head = head + len;
  return len;
}

uint32_t ringbuf_peek(const ringbuf_t *rb, const uint8_t **ptr)
{
  uint32_t used = rb->head - rb->tail;
  uint32_t idx  = rb->tail & rb->mask;
  uint32_t n    = rb->size - idx;         /* contiguous up to buffer end */
  if (n > used) n = used;
  RB_FENCE();                             /* see data written before head */
  *ptr = &rb->buf[idx];
  return n;
}

void ringbuf_consume(ringbuf_t *rb, uint32_t n)
{
  uint32_t used = rb->head - rb->tail;
  if (n > used) n = used;
  RB_FENCE();                             /* finish reading before freeing */
  rb->tail += n;
}
```

### Build error hit and fixed

The first build failed in `usbd_cdc_if.c`:
```
usbd_cdc_if.c:376:20: error: 'tx_head' undeclared here (not in a function); did you mean 'head'?
usbd_cdc_if.c:377:27: error: 'USB_TX_RING_MASK' undeclared here (not in a function)
usbd_cdc_if.c:101:27: error: initializer element is not constant
... (more errors about __DMB, NVIC_DisableIRQ, 'data', 'len')
```
Cause: when the new `usb_write` was pasted, the **old `usb_write` body was left below it** (lines 376–390). Those lines sat outside any function, so every error came from them. The "initializer element is not constant" at line 101 came from the same leftover: line 378 (`USB_TX_RING_SIZE - idx`) was at file scope, and the compiler reported it at the macro's line.
Fix: delete the leftover lines, from `uint32_t head  = tx_head;` down to the stray `}` just before `int usb_printf(...)`. After that, the new `usb_write` ends with `return len; }` and `usb_printf` follows directly.

Rebuilt, flashed, and checked on COM41. **Result: builds and works.** Tick, echo, `b` burst, and USB behave the same as before.
